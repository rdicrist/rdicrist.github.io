// Method 2: Grab image hashes from ALL elements in the page (including hidden/lazy ones)
const html = document.documentElement.innerHTML;
const hashes = [...html.matchAll(/i\.imgur\.com\/([A-Za-z0-9]+)\.\w+/g)]
  .map(m => m[1]);

const images = [...new Set(hashes)].map(h => `https://i.imgur.com/${h}`);

console.log(`Found ${images.length} images`);
console.log(JSON.stringify(images, null, 2));
